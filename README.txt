# 1) Компиляция — создаёт исполняемый файл ./csma_sim
g++ -std=c++17 csma_sim.cpp -O2 -o csma_sim
# (комментарий) Компилируем C++ исходник в исполняемый файл.

# 2) Быстрый тест одиночного запуска — вывод в консоль 
./csma_sim --N 10 --alpha 50 --simtime 10
# (комментарий) Запуск симуляции с N=10, alpha=50, время 10s; проверяем, работает ли всё и выводит метрики.

# 3) Тест с CSV — проверяем, что параметр --csv теперь работает
./csma_sim --N 10 --alpha 50 --simtime 10 --csv results.csv
# (комментарий) То же, но добавляет строку в results.csv. Если файл не существует, он создастся с заголовком.

# 4) Удаляем старый CSV (если хотим начать с нуля)
rm -f results.csv
# (комментарий) Удаляет файл results.csv без запроса подтверждения (-f). 

# 5) Массовый прогон: для alpha = 10,50,100 и N = 10,20,...,200, сохраняем в results.csv
for alpha in 10 50 100; do
    for N in $(seq 10 10 200); do
        ./csma_sim --N $N --alpha $alpha --simtime 10 --csv results.csv
    done
done
# (комментарий) Собираем таблицу результатов для трёх значений alpha и ряда N; всё сохраняется в results.csv.

# 6) Построение графиков (предварительно убедись, что pandas и matplotlib установлены)
pip install --user pandas matplotlib
# (комментарий) Устанавливаем библиотеки Python, если ещё не установлены. (--user ставит в локальный профиль пользователя)

# 7) Запуск скрипта, который строит графики из results.csv
python3 plot_results.py
# (комментарий) Считывает results.csv и создаёт файлы channel_util_vs_N.png, max_queue_vs_N.png, loss_vs_N.png

# 8) Поиск максимального N при channel_util <= 0.5 (пример для alpha=50)
python3 find_max_n.py 50
# (комментарий) Выполняет бинарный поиск, многократно вызывая ./csma_sim, и печатает найденный N и util.

# 9) Поиск максимального N при котором очередь считается стабильной (пример alpha=50)
python3 find_stable_queue.py 50
# (комментарий) Проверяет стабильность очереди, сравнивая max_queue при T1=20 и T2=40, и ищет максимальное N.
